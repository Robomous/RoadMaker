# test_fill_predicates.cpp pins the surface-fill point-in-polygon predicate
# (#442), which lives in the internal, non-installed core/src/mesh/fill_backend.hpp
# — the defect is invisible through the public mesh API. The header is entirely
# `inline`, so including it from a test TU is ODR-safe.
target_include_directories(roadmaker_core_tests PRIVATE
  ${CMAKE_SOURCE_DIR}/core/src
  # test_osm_scale.cpp shares the bench's district generator, so the scale
  # path is compiled and exercised on every platform rather than only where
  # the bench runs.
  ${CMAKE_CURRENT_SOURCE_DIR})

endif()

  RM_FUZZ_CORPUS_XOSC_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fuzz/corpus_xosc"
  # Committed GIS fixtures (p7-s2, #242), generated by
  # scripts/gen_gis_fixtures.py. Read off disk on purpose: a format reader
  # tested only against bytes the same test just built agrees with itself
  # about everything, including its mistakes.
  RM_GIS_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/gis"
  # Committed LAS/LAZ fixtures (p7-s3, #243), generated by
  # scripts/gen_lidar_fixtures.py — except amsterdam_tile.laz, which needs
  # the codec and comes from the rm_make_laz_fixture tool below.
  RM_LIDAR_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/lidar"
  RM_OSM_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/osm"
  # Committed glTF/GLB fixtures (p6-s8, #322), generated by
  # scripts/gen_gltf_fixtures.py — including the malformed ones, which are
  # what #322's fuzz-adjacent acceptance criterion actually asks for.
  RM_GLTF_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/gltf"
  # test_rm_registry.cpp scans the committed reader/writer sources and the
  # sibling tests for rm: code coverage (fmt-s2, #326) — the same
  # read-the-repo-off-disk move as RM_DOCS_DIR above.
  RM_CORE_SRC_DIR="${CMAKE_SOURCE_DIR}/core/src"
  RM_CORE_TESTS_DIR="${CMAKE_CURRENT_SOURCE_DIR}"

# The scale harness (p7-s4, #244) — the measurement half, discharging #54.
#

# gtest_discover_tests: `ctest -j 4` runs some 1 800 processes on a four-core
# runner, and a timing test inside that set measures the scheduler rather than
# the code. Its own CI job runs it serially, alone, on ubuntu-latest.
#
# core/tests/test_osm_scale.cpp — which IS in the normal suite above — builds
# the same district at reduced size and asserts CORRECTNESS only, so the scale
# path compiles and is exercised on all three platforms every PR while only
# the numbers are Linux-gated.

  scale/synthetic_district.cpp)

  GTest::gtest
